# WinDbg Output-Mode Filter Commands (Prometheus-XIM)
# Use these when normal breakpoints are too noisy and you only want
# specific controller output transitions.

# =============================================================================
# WHY THESE FILTERS WERE USED
# =============================================================================
# Reason 1: WriteValueInternal is hit by many BLE writes, not only CFG1D settings.
# Reason 2: We needed only type=0x15, subtype=0x1D, length=0x64 payloads.
# Reason 3: To verify output mapping quickly, we filtered by mode byte @ +0x5E.
# Reason 4: This avoids drowning in unrelated packets while toggling UI settings.

# =============================================================================
# BASE SETUP
# =============================================================================
.symfix
.sympath+ srv*C:\symbols*https://msdl.microsoft.com/download/symbols
.reload /f
bu Windows_Devices_Bluetooth!GattCharacteristic::WriteValueInternal
bc *

# Optional logging
.logopen /t c:\tmp\Prometheus-XIM\examples\captures\raw\windbg_mode_filters.log

# =============================================================================
# PACKET SHAPE FILTER (CFG1D only)
# =============================================================================
# Break only when:
#  - len == 0x64
#  - payload[0] == 0x15
#  - payload[1] == 0x1D

bp @!"Windows_Devices_Bluetooth!Windows::Devices::Bluetooth::GenericAttributeProfile::GattCharacteristic::WriteValueInternal" ".if (dwo(@rdx+0x58)==0x64) { .if (by(poi(@rdx+0x50))==0x15) { .if (by(poi(@rdx+0x50)+0x1)==0x1D) { .echo [CFG1D]; .printf \"mode=%02x rate=%02x comp=%02x\\n\", by(poi(@rdx+0x50)+0x5E), by(poi(@rdx+0x50)+0x5F), by(poi(@rdx+0x50)+0x6D); db poi(@rdx+0x50) L64; } } }; gc"

# Variant for builds/flows that emit subtype 0x00 instead of 0x1D (parser-safe nesting)
bp @!"Windows_Devices_Bluetooth!Windows::Devices::Bluetooth::GenericAttributeProfile::GattCharacteristic::WriteValueInternal" ".if (dwo(@rdx+0x58)==0x64) { .if (by(poi(@rdx+0x50))==0x15) { .if (by(poi(@rdx+0x50)+0x1)==0x1D) { .echo [CFG1D_VARIANT]; .printf \"subtype=%02x\\n\", by(poi(@rdx+0x50)+0x1); db poi(@rdx+0x50) L64; } .else { .if (by(poi(@rdx+0x50)+0x1)==0x00) { .echo [CFG1D_VARIANT]; .printf \"subtype=%02x\\n\", by(poi(@rdx+0x50)+0x1); db poi(@rdx+0x50) L64; } } } }; gc"
